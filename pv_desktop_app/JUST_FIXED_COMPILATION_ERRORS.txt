FIX #3: Double to Decimal Conversion
─────────────────────────────────────

File A: Services/DecisionCalculatorService.cs

Before:
    decimal moyAdmis = etudiants.Where(e => e.Decision == "Admis")
        .Average(e => (double)e.MoyenneGenerale);
    // ERROR: Average() returns double, can't assign to decimal implicitly

After:
    decimal moyAdmis = etudiants.Where(e => e.Decision == "Admis").Any()
        ? (decimal)etudiants.Where(e => e.Decision == "Admis")
            .Average(e => (double)e.MoyenneGenerale)
        : 0;
    // CORRECT: Explicit cast + safety check

Occurrences: 3 (lines 117, 119, 121)
Status: ✅ FIXED


File B: Services/ExcelDecisionExportService.cs

Before:
    decimal moyAdmis = etudiants.Where(e => e.Decision == "Admis")

After:
    decimal moyAdmis = etudiants.Where(e => e.Decision == "Admis").Any()

            .Average(e => e.MoyenneGenerale)
        : 0;
